    <!-- JavaScript Logic -->
    <script>
        // Initial Default Student List
        let students = [
            { regNo: "710023104001", name: "Aravind Swamy" },
            { regNo: "710023104002", name: "Bhavana K." },
            { regNo: "710023104003", name: "Dinesh Kumar" },
            { regNo: "710023104004", name: "Karpaga V." },
            { regNo: "710023104005", name: "Naveen Raj" }
        ];

        // Store Current Marks in RAM: { regNo: 'P' or 'A' }
        let currentAttendance = {};
        
        // Persistent Attendance History Logs
        let attendanceHistory = JSON.parse(localStorage.getItem('dept_attendance_logs')) || [];

        // Set default date to today
        document.getElementById('attDate').valueAsDate = new Date();

        function renderRoster() {
            const tbody = document.getElementById('attendanceRoster');
            tbody.innerHTML = '';
            
            students.forEach(std => {
                if (!currentAttendance[std.regNo]) {
                    currentAttendance[std.regNo] = 'P'; // Default Present
                }
                const isPresent = currentAttendance[std.regNo] === 'P';

                tbody.innerHTML += `
                    <tr class="hover:bg-slate-750">
                        <td class="p-4 font-mono text-slate-300">${std.regNo}</td>
                        <td class="p-4 font-semibold text-white">${std.name}</td>
                        <td class="p-4 text-center">
                            <div class="inline-flex rounded-md shadow-sm" role="group">
                                <button type="button" onclick="setStatus('${std.regNo}', 'P')" 
                                    class="px-4 py-1.5 text-xs font-bold rounded-l-lg transition ${isPresent ? 'bg-emerald-600 text-white' : 'bg-slate-700 text-slate-400 hover:bg-slate-600'}">
                                    PRESENT
                                </button>
                                <button type="button" onclick="setStatus('${std.regNo}', 'A')" 
                                    class="px-4 py-1.5 text-xs font-bold rounded-r-lg transition ${!isPresent ? 'bg-rose-600 text-white' : 'bg-slate-700 text-slate-400 hover:bg-slate-600'}">
                                    ABSENT
                                </button>
                            </div>
                        </td>
                    </tr>
                `;
            });

            updateSummaryBadge();
        }

        function setStatus(regNo, status) {
            currentAttendance[regNo] = status;
            renderRoster();
        }

        function markAll(status) {
            students.forEach(std => {
                currentAttendance[std.regNo] = status;
            });
            renderRoster();
        }

        function updateSummaryBadge() {
            const total = students.length;
            const presentCount = Object.values(currentAttendance).filter(v => v === 'P').length;
            document.getElementById('summaryBadge').innerText = `Present: ${presentCount} / ${total}`;
        }

        function saveAttendance() {
            const date = document.getElementById('attDate').value;
            const className = document.getElementById('attClass').value;
            const subject = document.getElementById('attSubject').value || 'General';

            if (!date) {
                alert("Please select a date.");
                return;
            }

            const total = students.length;
            const presentCount = Object.values(currentAttendance).filter(v => v === 'P').length;
            const percentage = ((presentCount / total) * 100).toFixed(1);

            const record = {
                date,
                className,
                subject,
                presentCount,
                total,
                percentage,
                details: { ...currentAttendance }
            };

            attendanceHistory.push(record);
            localStorage.setItem('dept_attendance_logs', JSON.stringify(attendanceHistory));
            
            alert(`Attendance saved successfully for ${className} (${subject})!`);
            switchTab('reports');
        }

        function renderReports() {
            const tbody = document.getElementById('reportsTable');
            tbody.innerHTML = '';

            if (attendanceHistory.length === 0) {
                tbody.innerHTML = `<tr><td colspan="5" class="p-4 text-center text-slate-500">No attendance records saved yet.</td></tr>`;
                return;
            }

            attendanceHistory.slice().reverse().forEach(rec => {
                tbody.innerHTML += `
                    <tr class="hover:bg-slate-750">
                        <td class="p-4 font-mono text-slate-300">${rec.date}</td>
                        <td class="p-4 font-semibold text-white">${rec.className}</td>
                        <td class="p-4 text-slate-300">${rec.subject}</td>
                        <td class="p-4 font-mono">${rec.presentCount} / ${rec.total}</td>
                        <td class="p-4 font-bold ${rec.percentage >= 75 ? 'text-emerald-400' : 'text-rose-400'}">${rec.percentage}%</td>
                    </tr>
                `;
            });
        }

        function handleAddStudent(e) {
            e.preventDefault();
            const regNo = document.getElementById('newRegNo').value.trim();
            const name = document.getElementById('newName').value.trim();

            if (regNo && name) {
                students.push({ regNo, name });
                currentAttendance[regNo] = 'P';
                document.getElementById('addStudentForm').reset();
                alert(`Added ${name} to the roster.`);
                switchTab('mark');
            }
        }

        function exportToCSV() {
            if (attendanceHistory.length === 0) {
                alert("No records available to export.");
                return;
            }

            let csvContent = "data:text/csv;charset=utf-8,Date,Class,Subject,Present,Total,Percentage\n";
            attendanceHistory.forEach(r => {
                csvContent += `${r.date},${r.className},${r.subject},${r.presentCount},${r.total},${r.percentage}%\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `Attendance_Report_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById(tabId + '-tab').classList.remove('hidden');
            if (tabId === 'mark') renderRoster();
            if (tabId === 'reports') renderReports();
        }

        // Initial Load
        renderRoster();
    </script>
